cmake_minimum_required(VERSION 3.20)
project(ChainMorningstarAssetExporter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "Disable nifly dependency tests" FORCE)
set(CMS_NIFLY_ROOT "" CACHE PATH "Source tree of nifly at cca0a770094bb962fb28ea1fec5ea903e68fda8e")
if(NOT EXISTS "${CMS_NIFLY_ROOT}/include/NifFile.hpp")
  message(FATAL_ERROR "Set CMS_NIFLY_ROOT to the documented, pinned nifly source tree.")
endif()
add_subdirectory("${CMS_NIFLY_ROOT}" "${CMAKE_BINARY_DIR}/nifly")
add_executable(export_reference_nif export_reference_nif.cpp)
target_link_libraries(export_reference_nif PRIVATE nifly)
